<script>
const $=id=>document.getElementById(id);
function fmt(n,d=3){return Number.isFinite(n)?n.toFixed(d):'—'}
async function loadLiveData(){
    const btn=$('liveBtn');
    btn.disabled=true;
    btn.textContent="⏳正在加载行情…";
    $('liveStatus').textContent="请求行情中，请等待";
    try{
        const fs=encodeURIComponent('m:0+t:6,m:0+t:80,m:0+t:81,m:1+t:2,m:1+t:23');
        const fields='f12,f14,f2,f3,f5,f6';
        const cbName='cb_'+Date.now()+'_'+Math.floor(Math.random()*99999);
        const url=`https://push2.eastmoney.com/api/qt/clist/get?pn=1&pz=6000&po=1&np=1&fltt=2&invt=2&fid=f3&fs=${fs}&fields=${encodeURIComponent(fields)}&cb=${cbName}`;
        await new Promise((resolve,reject)=>{
            const script=document.createElement('script');
            let done=false;
            const timer=setTimeout(()=>{if(!done){done=true;script.remove();reject(new Error("请求超时"));}},12000);
            window[cbName]=function(res){
                if(done) return; done=true; clearTimeout(timer); script.remove();
                if(res.rc!==0 || !res.data || !res.data.diff) return reject(new Error("接口返回异常"));
                let adv=0,dec=0,flat=0,upVol=0,downVol=0,amount=0,valid=0;
                for(const r of res.data.diff){
                    const ch=Number(r.f3),vol=Number(r.f5),amt=Number(r.f6);
                    if(!Number.isFinite(ch)) continue;
                    valid++;
                    if(ch>0){adv++;upVol+=Number.isFinite(vol)?vol:0;}
                    else if(ch<0){dec++;downVol+=Number.isFinite(vol)?vol:0;}
                    else flat++;
                    if(Number.isFinite(amt)) amount+=amt;
                }
                if(valid<1000) return reject(new Error("有效股票数量过少"));
                const total=adv+dec+flat;
                const upPct=adv/total*100;
                const downPct=dec/total*100;
                const flatPct=flat/total*100;
                const adRatio=adv/dec;
                const volRatio=upVol/downVol;
                const trin=adRatio/volRatio;
                $('advDisplay').textContent=adv;
                $('decDisplay').textContent=dec;
                $('upPct').textContent=upPct.toFixed(2)+'%';
                $('upPct2').textContent=upPct.toFixed(2)+'%';
                $('downPct').textContent=downPct.toFixed(2)+'%';
                $('flatPct').textContent=flatPct.toFixed(2)+'%';
                $('upBar').style.width=upPct+'%';
                $('downBar').style.width=downPct+'%';
                $('flatBar').style.width=flatPct+'%';
                $('trinCalc').textContent=fmt(trin);
                $('totalCnt').textContent=valid;
                $('totalAmt').textContent=(amount/100000000).toFixed(0);

                const cb2='cbidx_'+Date.now();
                const s=document.createElement('script');
                s.src=`https://push2.eastmoney.com/api/qt/stock/get?secid=1.000001&fields=f43,f170&cb=${cb2}`;
                window[cb2]=(idxRes)=>{
                    if(idxRes?.data){
                        const price=Number(idxRes.data.f43)/100;
                        const chg=Number(idxRes.data.f170)/100;
                        $('idx').textContent=price.toFixed(2);
                        $('idxNote').textContent=(chg>0?"+":"")+chg.toFixed(2)+"%";
                    }
                    delete window[cb2];
                    s.remove();
                };
                document.head.appendChild(s);

                $('sourceBadge').textContent="✅行情已接入";
                $('sourceBadge').style.color="#39d6a0";
                const time=new Date().toLocaleString('zh-CN',{hour12:false});
                $('liveStatus').textContent="更新时间："+time;

                let tip="";
                if(upPct<30) tip="⚠️市场广度极弱，普跌行情，TRIN<1也不能代表超卖";
                else if(upPct<50) tip="⚠️广度偏弱，反弹需要上涨家数继续放大确认";
                else tip="✅市场广度改善";
                $('summary').innerHTML=`<strong>${tip}</strong><br>上涨${adv}家，下跌${dec}家；上涨占比${upPct.toFixed(2)}%；TRIN=${trin.toFixed(3)}。`;
                resolve();
            };
            script.onerror=()=>{if(!done){done=true;clearTimeout(timer);script.remove();reject(new Error("网络加载失败"));}};
            document.head.appendChild(script);
        });
    }catch(err){
        $('liveStatus').textContent="❌失败："+err.message;
        $('sourceBadge').textContent="连接失败";
        $('sourceBadge').style.color="#ff6877";
        $('summary').innerText="获取行情失败，可能是网络、接口限流；过一会再点。";
    }finally{
        btn.disabled=false;
        btn.textContent="🔄 获取实时行情";
    }
}
$('liveBtn').addEventListener('click',loadLiveData);
</script>
